    // Nefelómetro Optoelectrónico (Esquema Real FA/FB) - FCFM BUAP
// FA (0° Transmitancia) -> R2(1.6K) + C1(10uF) + R3(680) + D2(BZX55C5V1) -> A0 (ADC_0)
// FB (90° Dispersión)   -> R1(1.6K) + C2(10uF) + R4(680) + D1(BZX55C5V1) -> A1 (ADC_90)
const int PIN_ADC_0  = A0;
const int PIN_ADC_90 = A1;
const float V_REF = 5.00;
const float V_DARK = 0.0012;

float readOversampledVolts(int pin) {
  unsigned long sum = 0;
  for (int i = 0; i < 64; i++) { sum += analogRead(pin); delayMicroseconds(250); }
  return ((sum / 16.0) / 4095.0) * V_REF;
}

void setup() {
  Serial.begin(115200);
  analogReference(DEFAULT);
}

void loop() {
  float v0  = readOversampledVolts(PIN_ADC_0);
  float v90 = readOversampledVolts(PIN_ADC_90);
  float r_ratio = (v90 - V_DARK) / max(0.001, (v0 - V_DARK));
  float ntu = max(0.0, (v90 - V_DARK) / 0.0268);
  Serial.print("{\"v90\":"); Serial.print(v90, 4);
  Serial.print(",\"v0\":");  Serial.print(v0, 4);
  Serial.print(",\"ratio\":"); Serial.print(r_ratio, 4);
  Serial.print(",\"ntu\":"); Serial.print(ntu, 2);
  Serial.println("}");
  delay(50);
}
